debounce.ts 
export function debounce<T extends unknown[]>(
  fn: (...args: T) => void,
  wait: number,
) {
  let t: ReturnType<typeof setTimeout>;
  return (...args: T) => {
    clearTimeout(t);
    t = setTimeout(() => fn(...args), wait);
  };
}
